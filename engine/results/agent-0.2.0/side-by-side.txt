side_by_side.py:

    """vpnw 0.1.0 and 0.2.0 side by side on one machine: start-up (the two
    versions run in turn, 60 times each, so both see the same load) and the
    resident memory of vpnw while a sealed program runs, split into anonymous
    memory and pages of the program file.
    
        python3 side_by_side.py vpnw-0.1.0 vpnw-0.2.0
    """
    import os
    import statistics
    import subprocess
    import sys
    import time
    
    ENV = {k: v for k, v in os.environ.items() if k.lower() not in ("http_proxy", "https_proxy", "all_proxy", "no_proxy")}
    ENV["NO_COLOR"] = "1"
    bins = sys.argv[1:3]
    
    
    def version(b):
        return subprocess.run([b, "version"], capture_output=True, text=True, env=ENV).stdout.split()[1]
    
    
    names = [version(b) for b in bins]
    print("host:", os.uname().release, os.uname().machine, "cpus:", os.cpu_count())
    for n, b in zip(names, bins):
        print("vpnw %s: %s, %d bytes" % (n, b, os.path.getsize(b)))
    print()
    
    times = {(n, be): [] for n in names for be in ("sealed", "env")}
    bare = []
    for _ in range(60):
        t = time.perf_counter()
        subprocess.run(["/bin/true"])
        bare.append((time.perf_counter() - t) * 1000)
        for n, b in zip(names, bins):
            for be in ("sealed", "env"):
                t = time.perf_counter()
                subprocess.run([b, "run", "--backend", be, "--no-save", "--", "/bin/true"], env=ENV,
                               stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
                times[(n, be)].append((time.perf_counter() - t) * 1000)
    print("start-up and exit of /bin/true, median of 60 runs, the versions in turn")
    print("  bare                  %6.2f ms" % statistics.median(bare))
    for (n, be), xs in times.items():
        print("  vpnw %s, %-7s %6.2f ms" % (n, be, statistics.median(xs)))
    print()
    
    
    def status(pid):
        out = {}
        for line in open("/proc/%d/status" % pid):
            k = line.split(":")[0]
            if k in ("VmRSS", "RssAnon", "RssFile"):
                out[k] = int(line.split()[1]) / 1024
        return out
    
    
    print("memory of vpnw while a sealed program runs, resident, 3 runs each")
    for n, b in zip(names, bins):
        for _ in range(3):
            p = subprocess.Popen([b, "run", "--backend", "sealed", "--no-save", "--", "/bin/sleep", "2"], env=ENV,
                                 stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            time.sleep(1.0)
            s = status(p.pid)
            print("  vpnw %s  %.1f MB: anonymous %.1f MB, pages of the program file %.1f MB" % (n, s["VmRSS"], s["RssAnon"], s["RssFile"]))
            p.wait()
